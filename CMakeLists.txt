cmake_minimum_required(VERSION 3.20)
project(Quarrylight VERSION 1.0 LANGUAGES C CXX)

if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QUARRYLIGHT_BUILD_APP "Build the desktop OpenGL application" ON)

include(FetchContent)
find_package(glm 1.0 CONFIG QUIET)
if(NOT TARGET glm::glm)
    set(GLM_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glm
        URL https://codeload.github.com/g-truc/glm/tar.gz/refs/tags/1.0.1
        URL_HASH SHA256=9f3174561fd26904b23f0db5e560971cbf9b3cbda0b280f04d5c379d03bf234c)
    FetchContent_MakeAvailable(glm)
endif()

function(quarrylight_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(quarrylight_world src/world.cpp src/campus.cpp src/mesher.cpp src/player.cpp)
target_include_directories(quarrylight_world PUBLIC src)
target_link_libraries(quarrylight_world PUBLIC glm::glm)
quarrylight_warnings(quarrylight_world)

if(QUARRYLIGHT_BUILD_APP)
    find_package(glfw3 3.3 CONFIG QUIET)
    if(NOT TARGET glfw)
        set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
        set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
        endif()
        FetchContent_Declare(glfw
            URL https://codeload.github.com/glfw/glfw/tar.gz/refs/tags/3.4
            URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
        FetchContent_MakeAvailable(glfw)
    endif()
    add_library(quarrylight_glad STATIC vendor/glad/src/gl.c)
    target_include_directories(quarrylight_glad PUBLIC vendor/glad/include)
    add_executable(quarrylight src/main.cpp src/renderer.cpp src/hud.cpp)
    target_link_libraries(quarrylight PRIVATE quarrylight_world quarrylight_glad glfw)
    target_compile_definitions(quarrylight PRIVATE GLFW_INCLUDE_NONE)
    quarrylight_warnings(quarrylight)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(quarrylight_tests tests/world_tests.cpp)
    target_link_libraries(quarrylight_tests PRIVATE quarrylight_world)
    quarrylight_warnings(quarrylight_tests)
    add_test(NAME quarrylight_world COMMAND quarrylight_tests)
endif()
